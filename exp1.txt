# Step 1: Import Required Libraries

import pandas as pd
import numpy as np
from sklearn.preprocessing import LabelEncoder


# Step 2: Load Dataset

df = pd.read_csv("titanic.csv")
# Display first 5 rows
df.head()


# Step 3: Check dimensions of dataset

print("Shape of dataset:", df.shape)
print("Number of rows:", df.shape[0])
print("Number of columns:", df.shape[1])

# Step 4: Basic Information

df.info()


# Step 5: Check Missing Values

print(df.isnull().sum())

# Step 6: Statistical Summary

df.describe()


# Step 7: Data Types of Variables

print(df.dtypes)


# Step 8: Fill Missing Values

df['Age'] = df['Age'].fillna(df['Age'].mean())

# Fill missing Age with mean
df['Age'].fillna(df['Age'].mean(), inplace=True)

# Fill missing Embarked with mode
df['Embarked'] = df['Embarked'].fillna(df['Embarked'].mode()[0])


# Fill missing Fare with mean if needed
df['Fare'] = df['Fare'].fillna(df['Fare'].mean())

print(df.isnull().sum())



# Step 9: Convert Data Types if Required
# Converting categorical text to numeric.

df['Age'] = df['Age'].astype(int)

print(df.dtypes)


# Step 10: Convert Categorical Variables into Quantitative Variables

label_encoder = LabelEncoder()

df['Sex'] = label_encoder.fit_transform(df['Sex'])
df['Embarked'] = label_encoder.fit_transform(df['Embarked'])

df.head()


# Final Dataset Check

df.info()
df.head()

Normalization : 
#Imports scaler for normalization.
from sklearn.preprocessing import MinMaxScaler

#Creates normalization object.
scaler = MinMaxScaler()

#Normalizes Age and Fare columns to values between 0 and 1.
df[['Age', 'Fare']] = scaler.fit_transform(df[['Age', 'Fare']])

---------------------------------------------------
Before:

Age	Fare
22	7.25
38	71.28

After:

Age	Fare
0.27	0.014
0.47	0.139


Because normalization is applied to numeric continuous columns.

Not categorical columns like:

Sex
Embarked
-------------------------------------------
Why this matters

Suppose before normalization:

Feature	Values
Age	18 to 80
Fare	7 to 512

See the issue?

Fare values are much bigger than Age.

Some machine learning algorithms may think:

Fare is more important just because numbers are larger.

Not because it actually is.

Normalization fixes this.
